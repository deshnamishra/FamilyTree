const PersonCard = ({ person, onSelect, isSelected, variant = 'default', scale = 1 }) => {
  const age = new Date().getFullYear() - person.birthYear;

  const getCardClasses = () => {
    let classes = "person-card";
    
    if (isSelected) {
      classes += " selected";
    }
    
    if (variant === 'selected') {
      classes += " selected-highlight";
    }
    
    return classes;
  };

  return (
    <div
      className={getCardClasses()}
      onClick={() => onSelect(person._id)}
      style={{ 
        transform: `scale(${scale})`,
        transition: 'all 0.4s cubic-bezier(0.4, 0, 0.2, 1)'
      }}
    >
      <div className="person-card-header">
        <div className="person-name">{person.name}</div>
        <div className="person-title">{person.title}</div>
      </div>
      <div className="person-details">
        <div>Born {person.birthYear}</div>
        <div>Age {age}</div>
      </div>
    </div>
  );
};

const TreeView = ({ selectedId, onSelect, familyData }) => {
  const [transitioning, setTransitioning] = useState(false);
  
  // Build family tree structure
  const treeStructure = useMemo(() => {
    const couples = new Map();
    const processed = new Set();
    const tree = [];
    
    // Helper function to find generation level
    const getGenerationLevel = (personId, visited = new Set()) => {
      if (visited.has(personId) || !familyData[personId]) return 0;
      
      visited.add(personId);
      const person = familyData[personId];
      
      if (!person.parentIds || person.parentIds.length === 0) {
        return 0; // Root generation
      }
      
      const parentLevels = person.parentIds.map(parentId => 
        getGenerationLevel(parentId, new Set(visited))
      );
      
      return Math.max(...parentLevels) + 1;
    };
    
    // Create couples and organize by generation
    Object.values(familyData).forEach(person => {
      if (processed.has(person._id)) return;
      
      const generation = getGenerationLevel(person._id);
      const coupleKey = person.partnerId ? 
        [person._id, person.partnerId].sort().join('-') : 
        person._id;
      
      if (!couples.has(coupleKey)) {
        const couple = {
          id: coupleKey,
          generation,
          members: [person],
          children: [],
          parents: person.parentIds || []
        };
        
        if (person.partnerId && familyData[person.partnerId]) {
          couple.members.push(familyData[person.partnerId]);
          processed.add(person.partnerId);
        }
        
        // Get all children of this couple
        const allChildren = new Set();
        couple.members.forEach(member => {
          if (member.childIds) {
            member.childIds.forEach(childId => allChildren.add(childId));
          }
        });
        
        couple.children = Array.from(allChildren).map(childId => familyData[childId]).filter(Boolean);
        
        couples.set(coupleKey, couple);
      }
      
      processed.add(person._id);
    });
    
    // Convert to array and sort by generation
    return Array.from(couples.values()).sort((a, b) => a.generation - b.generation);
  }, [familyData]);
  
  // Group by generation
  const generationGroups = useMemo(() => {
    const groups = {};
    treeStructure.forEach(couple => {
      if (!groups[couple.generation]) {
        groups[couple.generation] = [];
      }
      groups[couple.generation].push(couple);
    });
    return groups;
  }, [treeStructure]);

  const handleSelect = (personId) => {
    if (personId === selectedId) return;
    
    setTransitioning(true);
    
    setTimeout(() => {
      onSelect(personId);
      setTransitioning(false);
    }, 200);
  };

  const renderCouple = (couple) => {
    const isAnySelected = couple.members.some(member => member._id === selectedId);
    
    return (
      <div 
        key={couple.id}
        className={`couple-container ${isAnySelected ? 'couple-selected' : ''}`}
        style={{
          transform: transitioning && isAnySelected ? 'translateY(-5px)' : 'translateY(0)',
          opacity: transitioning && isAnySelected ? 0.8 : 1,
          transition: 'all 0.4s cubic-bezier(0.4, 0, 0.2, 1)'
        }}
      >
        <div className="couple-members">
          {couple.members.map((member, index) => {
            const isSelected = member._id === selectedId;
            const scale = isSelected ? 1.15 : 1;
            
            return (
              <div key={member._id} className="member-wrapper">
                <PersonCard
                  person={member}
                  onSelect={handleSelect}
                  isSelected={isSelected}
                  variant={isSelected ? 'selected' : 'default'}
                  scale={scale}
                />
                {index === 0 && couple.members.length > 1 && (
                  <div className="partnership-connector">
                    {/* <div className="heart-connector">💕</div> */}
                  </div>
                )}
              </div>
            );
          })}
        </div>
        
        {/* Children connector */}
        {couple.children.length > 0 && (
          <div className="children-connector">
            <div className="vertical-line"></div>
            <div className="horizontal-line"></div>
            <div className="children-branches">
              {couple.children.map((child, index) => (
                <div key={child._id} className="child-branch">
                  <div className="branch-line"></div>
                </div>
              ))}
            </div>
          </div>
        )}
      </div>
    );
  };

  const renderChildren = (couple) => {
    if (couple.children.length === 0) return null;
    
    return (
      <div key={`children-${couple.id}`} className="children-row">
        {couple.children.map(child => {
          const isSelected = child._id === selectedId;
          const scale = isSelected ? 1.15 : 1;
          
          return (
            <div key={child._id} className="child-container">
              <PersonCard
                person={child}
                onSelect={handleSelect}
                isSelected={isSelected}
                variant={isSelected ? 'selected' : 'default'}
                scale={scale}
              />
            </div>
          );
        })}
      </div>
    );
  };

  const sortedGenerations = Object.keys(generationGroups).map(Number).sort((a, b) => a - b);

  return (
    <div className="hierarchical-tree-view">
      <div className="tree-header">
        <h2 className="tree-title">Family Tree</h2>
        <p className="tree-subtitle">Click any family member to focus on them</p>
      </div>
      
      <div className="tree-container">
        {sortedGenerations.map(generation => (
          <div key={generation} className="generation-level">
            <div className="generation-header">
              <span className="generation-label">Generation {generation}</span>
            </div>
            
            <div className="couples-row">
              {generationGroups[generation].map(couple => renderCouple(couple))}
            </div>
            
            {/* Render children in the next level */}
            <div className="children-level">
              {generationGroups[generation].map(couple => renderChildren(couple))}
            </div>
          </div>
        ))}
      </div>
      
      {selectedId && (
        <div className="selected-member-info">
          <div className="info-panel">
            <h3 className="info-title">👤 {familyData[selectedId]?.name}</h3>
            <div className="member-details">
              <div className="detail-item">
                <span className="detail-label">Title:</span>
                <span className="detail-value">{familyData[selectedId]?.title || 'No title'}</span>
              </div>
              <div className="detail-item">
                <span className="detail-label">Age:</span>
                <span className="detail-value">{new Date().getFullYear() - familyData[selectedId]?.birthYear}</span>
              </div>
              <div className="detail-item">
                <span className="detail-label">Children:</span>
                <span className="detail-value">{familyData[selectedId]?.childIds?.length || 0}</span>
              </div>
              {familyData[selectedId]?.partnerId && (
                <div className="detail-item">
                  <span className="detail-label">Partner:</span>
                  <span className="detail-value">{familyData[familyData[selectedId].partnerId]?.name}</span>
                </div>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

function TreePage() {
  const [familyData, setFamilyData] = useState({});
  const [selectedId, setSelectedId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const extractBirthYear = (dateOfBirth) => {
    if (!dateOfBirth) return new Date().getFullYear() - 25;
    const date = new Date(dateOfBirth);
    return date.getFullYear();
  };

  const transformMemberData = (member) => {
    return {
      _id: member._id,
      name: member.name || "Unknown",
      title: member.title || "",
      gender: member.gender || "other",
      birthYear: extractBirthYear(member.dateOfBirth),
      dateOfBirth: member.dateOfBirth,
      parentIds: Array.isArray(member.parents) ? member.parents : [],
      childIds: Array.isArray(member.children) ? member.children : [],
      partnerId: member.partner || null,
      parents: Array.isArray(member.parents) ? member.parents : [],
      children: Array.isArray(member.children) ? member.children : [],
      partner: member.partner || null,
      createdAt: member.createdAt
    };
  };

  useEffect(() => {
    const loadData = async () => {
      try {
        setLoading(true);
        setError(null);
        
        const response = await mockFamilyService.fetchFamilyMembers();
        const members = response.data || [];

        const mapped = members.reduce((acc, member) => {
          if (!member._id) return acc;
          return {
            ...acc,
            [member._id]: transformMemberData(member)
          };
        }, {});

        setFamilyData(mapped);
        const firstValidId = Object.keys(mapped)[0] || null;
        setSelectedId(firstValidId);
      } catch (err) {
        console.error("Error loading family data:", err);
        setError(err.message || "Failed to load family data");
      } finally {
        setLoading(false);
      }
    };

    loadData();
  }, []);

  const handleSelect = (id) => {
    if (familyData[id]) setSelectedId(id);
  };

  if (loading) {
    return (
      <div className="loading-container">
        <div className="loading-spinner"></div>
        <p>Loading family tree...</p>
      </div>
    );
  }

  if (error) {
    return (
      <div className="error-container">
        <h2 className="error-title">Error loading family tree</h2>
        <p className="error-message">{error}</p>
        <button 
          onClick={() => window.location.reload()}
          className="error-retry-button"
        >
          Try Again
        </button>
      </div>
    );
  }

  return (
    <div className="tree-page">
      <header className="page-header">
        <div className="header-content">
          <h1 className="header-title">Family Tree</h1>
          {selectedId && familyData[selectedId] && (
            <div className="viewing-info">
              Currently viewing: <strong className="current-member-name">{familyData[selectedId].name}</strong>
              {familyData[selectedId].title && <span> ({familyData[selectedId].title})</span>}
            </div>
          )}
        </div>
      </header>

      <main className="main-content">
        <TreeView 
          selectedId={selectedId} 
          onSelect={handleSelect} 
          familyData={familyData} 
        />
      </main>
    </div>
  );
}

export default TreePage;
